<!-- wp:paragraph -->
<p>This is a list of all the features that are currently in MAIRA.</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">General Features</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Ability to configure settings per-car and per-track</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Ability to control most settings to various inputs (keys, knobs, buttons, hats)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>SimHub plugin</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Awesome force feedback experience - the main feature!</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>360 Hz (upscaled to 500 Hz) force feedback for all wheelbases</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Works for all types of wheelbases - gear driven, belt driven, and direct drive</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Adjust the main weight separately from the high frequency details</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Multiple FFB algorithms to choose from</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Prediction removes up to 2 frames of latency (1/30s)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel LFE</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Crash and curb protection</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Reduce forces and auto-center wheel when parked or not on track</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Soft clipping</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Vibrate on gear change and ABS activation</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Steering effects</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants steering effects</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Calibrated to each car</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel vibration, constant force, and pedal vibration effects</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Overlay to show how your car is behaving, similar to the skid-slip indicator on aircraft</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Pedal effects</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Works with all Simagic pedal haptic devices natively</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Works with some other brands via SimHub</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Mix, match, and tune different effects including -<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Gear change (separate settings for shift into gear and shift into neutral)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>ABS activation</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>RPM (constant or vibrate when its time to shift)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel lock and spin</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Clutch slip</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list --></li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Wind simulation</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Works with the custom MAIRA <a href="https://mairapp.com/home/maira-typhoon-wind-kit/" data-type="page" data-id="72">Typhoon Wind Kit</a></li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Equalizer-style fine tuning of fan speeds based on car speed</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wind curving</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Minimum fan speed (for those hot days)</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Sounds</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Custom sounds support (drop in your own wav files)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Adjust loudness and pitch (frequency shift)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Sound effects for -<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>ABS activation</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel lock and spin</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list --></li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Speech-to-Text</p>
<!-- /wp:paragraph -->

<!-- wp:list -->

<li>Uses the Google Chrome STT engine to convert voice chat messages to text</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Works with any language</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Text is shown in an overlay window</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Trading Paints</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Downloads paints from Trading Paints</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Notifies iRacing after paints have been downloaded (so cars are reloaded)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>No need to run the Trading Paints app</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->
