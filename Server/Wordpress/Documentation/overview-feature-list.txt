<!-- wp:paragraph -->
<p>This is a list of all the features that are currently in MAIRA.</p>
<!-- /wp:paragraph -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">General Features</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Ability to configure settings per-car and per-track</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Ability to control most settings to various inputs (keys, knobs, buttons, hats)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Controller profiles - save and switch between named sets of button/input mappings</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>SimHub plugin</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Awesome force feedback experience - the main feature!</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>360 Hz (upscaled to 500 Hz) force feedback for all wheelbases</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Works for all types of wheelbases - gear driven, belt driven, and direct drive</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Adjust the main weight separately from the high frequency details</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Multiple FFB algorithms to choose from</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Prediction removes up to 2 frames of latency (1/30s)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel LFE</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Crash and curb protection</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Reduce forces and auto-center wheel when parked or not on track</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Soft clipping</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Vibrate on gear change and ABS activation</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Steering effects</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants steering effects</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Calibrated to each car</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel vibration, constant force, and pedal vibration effects</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Overlay to show how your car is behaving, similar to the skid-slip indicator on aircraft</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Robot driver - outputs steering, brake, throttle, and shift up/down to a vJoy virtual joystick, letting MAIRA drive the car on track to learn and measure it</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Pedal effects</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Works with all Simagic pedal haptic devices natively</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Works with some other brands via SimHub</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Mix, match, and tune different effects including -<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Gear change (separate settings for shift into gear and shift into neutral)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>ABS activation</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>RPM (constant or vibrate when its time to shift)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel lock and spin</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Clutch slip</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list --></li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Wind simulation</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Works with the custom MAIRA <a href="https://mairapp.com/home/maira-typhoon-wind/" data-type="page" data-id="72">Typhoon Wind</a></li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Equalizer-style fine tuning of fan speeds based on car speed</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wind curving</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Minimum fan speed (for those hot days)</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">G Tensioner</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>USB serial-controlled <a href="https://mairapp.com/home/maira-g-tensioner/" data-type="page" data-id="647">G Tensioner</a> integration</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Real-time graphs showing surge, sway, and heave forces</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Responds to G-forces (surge, sway, heave) during racing</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Seat-of-pants effect for enhanced immersion</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Vibration effects (ABS, wheel slip, rumble)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Per-axis control (normal/inverted/disabled modes)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Separate left and right shoulder control</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Sounds</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Custom sounds support (drop in your own wav files)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Adjust loudness and pitch (frequency shift)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Sound effects for -<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>ABS activation</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Wheel lock and spin</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Understeer, oversteer, and seat-of-pants</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Brake and throttle warning (when both are pressed simultaneously)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>FFB clipping warning</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list --></li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Captions</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Uses ElevenLabs Speech-to-Text to convert iRacing voice chat into captions</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Runs in real time while radio traffic is active</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Text is shown in a movable/scalable overlay window</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Commentary</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>AI-powered race commentary using ElevenLabs Text-to-Speech</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Spotter calls for car proximity (car left, car right, three-wide situations, clear)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Flag announcements (green, yellow, white, checkered, blue, black, debris, etc.)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Multi-language support with customizable phrase templates</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Priority-based voice queue system</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Configurable reminder intervals for proximity calls</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Trading Paints</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Downloads paints from Trading Paints</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Notifies iRacing after paints have been downloaded (so cars are reloaded)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>No need to run the Trading Paints app</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Optionally delete downloaded paints (keeping your own) when iRacing closes, to keep your paint folder from filling up</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Graph</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Shows real time data graphs for -<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Input torque from iRacing (360 Hz and 60 Hz)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Input wheel LFE from iRacing</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Output torque from MAIRA</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Clutch, brake, and throttle pedal haptics</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Timer jitter</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list --></li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<!-- wp:paragraph {"className":"section"} -->
<p class="section">Overlays</p>
<!-- /wp:paragraph -->

<!-- wp:list -->
<ul class="wp-block-list"><!-- wp:list-item -->
<li>Gap monitor - movable/scalable overlay displaying real-time gap information to cars ahead and behind</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Delta monitor - movable/scalable overlay showing lap time and delta versus your last lap and versus the leader</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>Grip-O-Meter - movable/scalable overlay showing real-time grip usage (under/oversteer and seat-of-pants forces)</li>
<!-- /wp:list-item -->

<!-- wp:list-item -->
<li>All overlays can be made visible and draggable, optionally shown when off track and in replay mode</li>
<!-- /wp:list-item --></ul>
<!-- /wp:list -->

<p class="section">Game bridge</p>
<!-- /wp:paragraph -->
